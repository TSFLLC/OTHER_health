/*========================================================
  ONE-TIME STATIC MASKING FOR EXISTING DATA IN DEV
  TABLE: dbo.Appointment
  KEY  : AppointmentRowID  (change if needed)
=========================================================*/

UPDATE A
SET
    -- Names
    PatientFirstName = MF.MaskedFirstName,
    PatientLastName  = MF.MaskedLastName,

    -- MRN & IDs
    MRN              = MF.MaskedMRN,
    AppointmentID    = MF.MaskedAppointmentID,
    VisitEncounterID = MF.MaskedVisitEncounterID,
    EpicEncountersID = MF.MaskedEpicEncountersID,
    EpicAppointmentsID = MF.MaskedEpicAppointmentsID,
    ReferralCode     = MF.MaskedReferralCode,

    -- Address
    PatientHomeAddress1      = MF.MaskedAddress1,
    PatientHomeAddress2      = MF.MaskedAddress2,
    PatientHomeAddressCity   = MF.MaskedCity,
    PatientHomeAddressZipcode= MF.MaskedZip,

    -- Phones
    HomePhone   = MF.MaskedHomePhone,
    WorkPhone   = MF.MaskedWorkPhone,
    MobilePhone = MF.MaskedMobilePhone,

    -- Email
    PatientEmailAddress = MF.MaskedEmail,

    -- DOB
    PatientDateOfBirth = MF.MaskedDOB,

    -- Visit / Appointment clinical context
    VisitReasonCode        = MF.MaskedVisitReasonCode,
    VisitReasonDescription = MF.MaskedVisitReasonDesc,
    AppointmentDescription = MF.MaskedApptDesc,
    AppointmentTypeCode    = MF.MaskedApptType,
    AppointmentComment     = MF.MaskedApptComment,

    -- Insurance
    PrimaryHealthPlanName   = MF.MaskedPrimaryPlan,
    SecondaryHealthPlanName = MF.MaskedSecondaryPlan,

    -- Dates / Times
    ReferralCreatedOn      = MF.MaskedReferralCreatedOn,
    EncounterDate          = MF.MaskedEncounterDate,
    AppointmentDate        = MF.MaskedAppointmentDate,
    AppointmentStartTime   = MF.MaskedStartTime,
    AppointmentEndTime     = MF.MaskedEndTime
FROM dbo.Appointment AS A
CROSS APPLY
(
    -- Deterministic seed per row (same row => same masked values)
    SELECT ABS(CHECKSUM(A.AppointmentRowID)) AS Seed
) AS S
CROSS APPLY
(
    SELECT
        -- Simple deterministic pick from a small set of fake names
        CASE S.Seed % 10
            WHEN 0 THEN 'James'
            WHEN 1 THEN 'Mary'
            WHEN 2 THEN 'Robert'
            WHEN 3 THEN 'Patricia'
            WHEN 4 THEN 'John'
            WHEN 5 THEN 'Jennifer'
            WHEN 6 THEN 'Michael'
            WHEN 7 THEN 'Linda'
            WHEN 8 THEN 'David'
            ELSE 'Sarah'
        END AS MaskedFirstName,

        CASE (S.Seed / 10) % 10
            WHEN 0 THEN 'Smith'
            WHEN 1 THEN 'Johnson'
            WHEN 2 THEN 'Brown'
            WHEN 3 THEN 'Williams'
            WHEN 4 THEN 'Jones'
            WHEN 5 THEN 'Garcia'
            WHEN 6 THEN 'Miller'
            WHEN 7 THEN 'Davis'
            WHEN 8 THEN 'Lopez'
            ELSE 'Taylor'
        END AS MaskedLastName,

        -- MRN & IDs: pseudo-token (same length-ish, not real)
        RIGHT('0000000000' + CAST(ABS(CHECKSUM(A.MRN, S.Seed)) AS VARCHAR(10)), 10) AS MaskedMRN,
        RIGHT('APPT' + CAST(ABS(CHECKSUM(A.AppointmentID, S.Seed)) AS VARCHAR(10)), 10) AS MaskedAppointmentID,
        RIGHT('ENC' + CAST(ABS(CHECKSUM(A.VisitEncounterID, S.Seed)) AS VARCHAR(10)), 10) AS MaskedVisitEncounterID,
        RIGHT('EENC' + CAST(ABS(CHECKSUM(A.EpicEncountersID, S.Seed)) AS VARCHAR(10)), 10) AS MaskedEpicEncountersID,
        RIGHT('EAPT' + CAST(ABS(CHECKSUM(A.EpicAppointmentsID, S.Seed)) AS VARCHAR(10)), 10) AS MaskedEpicAppointmentsID,
        RIGHT('REF' + CAST(ABS(CHECKSUM(A.ReferralCode, S.Seed)) AS VARCHAR(10)), 10) AS MaskedReferralCode,

        -- Address
        CASE S.Seed % 5
            WHEN 0 THEN '123 Main St'
            WHEN 1 THEN '456 Oak Ave'
            WHEN 2 THEN '789 Pine Road'
            WHEN 3 THEN '321 Maple Drive'
            ELSE '654 Elm Street'
        END AS MaskedAddress1,

        NULL AS MaskedAddress2,  -- or 'Apt ' + CAST((S.Seed % 50)+1 AS VARCHAR(3))

        CASE (S.Seed / 5) % 5
            WHEN 0 THEN 'Houston'
            WHEN 1 THEN 'Dallas'
            WHEN 2 THEN 'Austin'
            WHEN 3 THEN 'San Antonio'
            ELSE 'Fort Worth'
        END AS MaskedCity,

        RIGHT('00000' + CAST(10000 + (S.Seed % 90000) AS VARCHAR(5)), 5) AS MaskedZip,

        -- Phones: 555-xxx-xxxx
        '555-' +
        RIGHT('000' + CAST((S.Seed % 800) + 100 AS VARCHAR(3)), 3) + '-' +
        RIGHT('0000' + CAST((S.Seed / 10) % 10000 AS VARCHAR(4)), 4) AS MaskedHomePhone,

        '555-' +
        RIGHT('000' + CAST((S.Seed / 3 % 800) + 100 AS VARCHAR(3)), 3) + '-' +
        RIGHT('0000' + CAST((S.Seed / 7) % 10000 AS VARCHAR(4)), 4) AS MaskedWorkPhone,

        '555-' +
        RIGHT('000' + CAST((S.Seed / 5 % 800) + 100 AS VARCHAR(3)), 3) + '-' +
        RIGHT('0000' + CAST((S.Seed / 11) % 10000 AS VARCHAR(4)), 4) AS MaskedMobilePhone,

        -- Email (fake domain)
        LOWER(
            CASE S.Seed % 10
                WHEN 0 THEN 'james'
                WHEN 1 THEN 'mary'
                WHEN 2 THEN 'robert'
                WHEN 3 THEN 'patricia'
                WHEN 4 THEN 'john'
                WHEN 5 THEN 'jennifer'
                WHEN 6 THEN 'michael'
                WHEN 7 THEN 'linda'
                WHEN 8 THEN 'david'
                ELSE 'sarah'
            END + '.' +
            CASE (S.Seed / 10) % 10
                WHEN 0 THEN 'smith'
                WHEN 1 THEN 'johnson'
                WHEN 2 THEN 'brown'
                WHEN 3 THEN 'williams'
                WHEN 4 THEN 'jones'
                WHEN 5 THEN 'garcia'
                WHEN 6 THEN 'miller'
                WHEN 7 THEN 'davis'
                WHEN 8 THEN 'lopez'
                ELSE 'taylor'
            END + '@examplemail.com'
        ) AS MaskedEmail,

        -- DOB: between 1940 and 1999, random month/day
        DATEFROMPARTS(
            1940 + (S.Seed % 60),
            1 + ((S.Seed / 100) % 12),
            1 + ((S.Seed / 1000) % 28)
        ) AS MaskedDOB,

        -- Visit reason / descriptions (generic text)
        'R' + RIGHT('0000' + CAST(S.Seed % 10000 AS VARCHAR(4)), 4) AS MaskedVisitReasonCode,
        'General consultation (masked)' AS MaskedVisitReasonDesc,
        'Routine visit (masked)' AS MaskedApptDesc,
        'ROUTINE' AS MaskedApptType,
        'Masked clinical comment' AS MaskedApptComment,

        -- Insurance plan names (generic)
        'Health Plan A (masked)' AS MaskedPrimaryPlan,
        'Health Plan B (masked)' AS MaskedSecondaryPlan,

        -- Dates: shift original dates by deterministic offset, or if NULL use synthetic
        DATEADD(DAY, (S.Seed % 365), ISNULL(A.ReferralCreatedOn, DATEFROMPARTS(2018,1,1))) AS MaskedReferralCreatedOn,
        DATEADD(DAY, (S.Seed % 365), ISNULL(A.EncounterDate,      DATEFROMPARTS(2018,1,1))) AS MaskedEncounterDate,
        DATEADD(DAY, (S.Seed % 365), ISNULL(A.AppointmentDate,    DATEFROMPARTS(2018,1,1))) AS MaskedAppointmentDate,

        -- Times: always within business hours 08:00–17:00
        TIMEFROMPARTS(8 + (S.Seed % 9), (S.Seed / 10) % 60, 0, 0, 0) AS MaskedStartTime,
        TIMEFROMPARTS(9 + (S.Seed % 9), (S.Seed / 20) % 60, 0, 0, 0) AS MaskedEndTime
) AS MF;

  TABLE: dbo.Appointment

    DROP TRIGGER dbo.trg_Appointment_Masking;
