/*========================================================
  ONE-TIME STATIC MASKING FOR DEV
  TABLE : S_cdpdynamics.CDP_Appointments
  SEED  : appointmentID
=========================================================*/

UPDATE A
SET
    -- Names
    PatientFirstName = M.MaskedFirstName,
    PatientLastName  = M.MaskedLastName,

    -- MRN & IDs
    MRN                    = M.MaskedMRN,
    [appointmentID]        = M.MaskedAppointmentID,   -- if you DON'T want to mask PK, comment this line
    VisitEncounterID       = M.MaskedVisitEncounterID,
    [Visit/Epic Encounters ID] = M.MaskedEpicEncountersID,
    [Epic Appointments ID]     = M.MaskedEpicAppointmentsID,
    [Referral Code]        = M.MaskedReferralCode,

    -- Address
    PatientHomeAddress1      = M.MaskedAddress1,
    PatientHomeAddress2      = M.MaskedAddress2,
    PatientHomeAddressCity   = M.MaskedCity,
    PatientHomeAddressZipcode= M.MaskedZip,

    -- Phones
    HomePhone   = M.MaskedHomePhone,
    WorkPhone   = M.MaskedWorkPhone,
    MobilePhone = M.MaskedMobilePhone,

    -- Email
    PatientEmailAddress = M.MaskedEmail,

    -- DOB
    PatientDateOfBirth = M.MaskedDOB,

    -- Visit / Appointment clinical fields
    VisitReasonCode        = M.MaskedVisitReasonCode,
    VisitReasonDescription = M.MaskedVisitReasonDesc,
    AppointmentDescription = M.MaskedApptDesc,
    AppointmentTypeCode    = M.MaskedApptType,
    AppointmentComment     = M.MaskedApptComment,

    -- Insurance
    PrimaryHealthPlanName   = M.MaskedPrimaryPlan,
    SecondaryHealthPlanName = M.MaskedSecondaryPlan,

    -- Dates / Times
    [Referral Createdon date] = M.MaskedReferralCreatedOn,
    EncounterDate             = M.MaskedEncounterDate,
    AppointmentDate           = M.MaskedAppointmentDate,
    AppointmentStartTime      = M.MaskedStartTime,
    AppointmentEndTime        = M.MaskedEndTime
FROM S_cdpdynamics.CDP_Appointments AS A
CROSS APPLY
(
    -- Deterministic seed per row using appointmentID
    SELECT ABS(CHECKSUM(A.[appointmentID])) AS Seed
) AS S
CROSS APPLY
(
    SELECT
        -- Masked first name
        CASE S.Seed % 10
            WHEN 0 THEN 'James'
            WHEN 1 THEN 'Mary'
            WHEN 2 THEN 'Robert'
            WHEN 3 THEN 'Patricia'
            WHEN 4 THEN 'John'
            WHEN 5 THEN 'Jennifer'
            WHEN 6 THEN 'Michael'
            WHEN 7 THEN 'Linda'
            WHEN 8 THEN 'David'
            ELSE 'Sarah'
        END AS MaskedFirstName,

        -- Masked last name
        CASE (S.Seed / 10) % 10
            WHEN 0 THEN 'Smith'
            WHEN 1 THEN 'Johnson'
            WHEN 2 THEN 'Brown'
            WHEN 3 THEN 'Williams'
            WHEN 4 THEN 'Jones'
            WHEN 5 THEN 'Garcia'
            WHEN 6 THEN 'Miller'
            WHEN 7 THEN 'Davis'
            WHEN 8 THEN 'Lopez'
            ELSE 'Taylor'
        END AS MaskedLastName,

        -- MRN & IDs: pseudo tokens (int values; SQL will cast to varchar if needed)
        ABS(CHECKSUM(A.MRN,                     S.Seed)) AS MaskedMRN,
        ABS(CHECKSUM(A.[appointmentID],         S.Seed)) AS MaskedAppointmentID,
        ABS(CHECKSUM(A.VisitEncounterID,        S.Seed)) AS MaskedVisitEncounterID,
        ABS(CHECKSUM(A.[Visit/Epic Encounters ID], S.Seed)) AS MaskedEpicEncountersID,
        ABS(CHECKSUM(A.[Epic Appointments ID],     S.Seed)) AS MaskedEpicAppointmentsID,
        ABS(CHECKSUM(A.[Referral Code],         S.Seed)) AS MaskedReferralCode,

        -- Address
        CASE S.Seed % 5
            WHEN 0 THEN '123 Main St'
            WHEN 1 THEN '456 Oak Ave'
            WHEN 2 THEN '789 Pine Road'
            WHEN 3 THEN '321 Maple Drive'
            ELSE '654 Elm Street'
        END AS MaskedAddress1,

        NULL AS MaskedAddress2,  -- or 'Apt ' + CAST((S.Seed % 50)+1 AS VARCHAR(3)),

        CASE (S.Seed / 5) % 5
            WHEN 0 THEN 'Houston'
            WHEN 1 THEN 'Dallas'
            WHEN 2 THEN 'Austin'
            WHEN 3 THEN 'San Antonio'
            ELSE 'Fort Worth'
        END AS MaskedCity,

        RIGHT('00000' + CAST(10000 + (S.Seed % 90000) AS VARCHAR(5)), 5) AS MaskedZip,

        -- Phones: 555-xxx-xxxx
        '555-' +
        RIGHT('000' + CAST((S.Seed % 800) + 100 AS VARCHAR(3)), 3) + '-' +
        RIGHT('0000' + CAST((S.Seed / 10) % 10000 AS VARCHAR(4)), 4) AS MaskedHomePhone,

        '555-' +
        RIGHT('000' + CAST((S.Seed / 3 % 800) + 100 AS VARCHAR(3)), 3) + '-' +
        RIGHT('0000' + CAST((S.Seed / 7) % 10000 AS VARCHAR(4)), 4) AS MaskedWorkPhone,

        '555-' +
        RIGHT('000' + CAST((S.Seed / 5 % 800) + 100 AS VARCHAR(3)), 3) + '-' +
        RIGHT('0000' + CAST((S.Seed / 11) % 10000 AS VARCHAR(4)), 4) AS MaskedMobilePhone,

        -- Email
        LOWER(
            CASE S.Seed % 10
                WHEN 0 THEN 'james'
                WHEN 1 THEN 'mary'
                WHEN 2 THEN 'robert'
                WHEN 3 THEN 'patricia'
                WHEN 4 THEN 'john'
                WHEN 5 THEN 'jennifer'
                WHEN 6 THEN 'michael'
                WHEN 7 THEN 'linda'
                WHEN 8 THEN 'david'
                ELSE 'sarah'
            END + '.' +
            CASE (S.Seed / 10) % 10
                WHEN 0 THEN 'smith'
                WHEN 1 THEN 'johnson'
                WHEN 2 THEN 'brown'
                WHEN 3 THEN 'williams'
                WHEN 4 THEN 'jones'
                WHEN 5 THEN 'garcia'
                WHEN 6 THEN 'miller'
                WHEN 7 THEN 'davis'
                WHEN 8 THEN 'lopez'
                ELSE 'taylor'
            END + '@examplemail.com'
        ) AS MaskedEmail,

        -- DOB: between 1940 and 1999
        DATEFROMPARTS(
            1940 + (S.Seed % 60),
            1 + ((S.Seed / 100) % 12),
            1 + ((S.Seed / 1000) % 28)
        ) AS MaskedDOB,

        -- Visit reason / descriptions
        'R' + RIGHT('0000' + CAST(S.Seed % 10000 AS VARCHAR(4)), 4) AS MaskedVisitReasonCode,
        'General consultation (masked)' AS MaskedVisitReasonDesc,
        'Routine visit (masked)' AS MaskedApptDesc,
        'ROUTINE' AS MaskedApptType,
        'Masked clinical comment' AS MaskedApptComment,

        -- Insurance plan names
        'Health Plan A (masked)' AS MaskedPrimaryPlan,
        'Health Plan B (masked)' AS MaskedSecondaryPlan,

        -- Dates: deterministically shifted
        DATEADD(DAY, (S.Seed % 365), ISNULL(A.[Referral Createdon date], DATEFROMPARTS(2018,1,1))) AS MaskedReferralCreatedOn,
        DATEADD(DAY, (S.Seed % 365), ISNULL(A.EncounterDate,           DATEFROMPARTS(2018,1,1))) AS MaskedEncounterDate,
        DATEADD(DAY, (S.Seed % 365), ISNULL(A.AppointmentDate,         DATEFROMPARTS(2018,1,1))) AS MaskedAppointmentDate,

        -- Times: synthetic within 08:00–17:00
        TIMEFROMPARTS(8 + (S.Seed % 9), (S.Seed / 10) % 60, 0, 0, 0) AS MaskedStartTime,
        TIMEFROMPARTS(9 + (S.Seed % 9), (S.Seed / 20) % 60, 0, 0, 0) AS MaskedEndTime
) AS M;
